-- =====================================================================
-- RPG TABLES: complete database setup
-- It creates everything in its final form: tables, security rules,
-- functions, triggers, image storage and default stats.

create schema if not exists private;


-- 1. TYPES ____________________________________________________________
create type public.member_role as enum ('dm', 'player');
create type public.member_status as enum ('invited', 'active');
create type public.dice_system as enum ('d6', 'd20');


-- 2. TABLES: Public profile for each user (username only) _____________

create table public.profiles (
  id uuid primary key references auth.users on delete cascade,
  username text not null unique check (char_length(username) between 3 and 30),
  created_at timestamptz not null default now()
);

create table public.games (
  id bigint generated always as identity primary key,
  name text not null,
  setting text, -- e.g. 'Star Wars', 'Middle-earth'
  description text, -- Markdown
  image_path text, -- path in the 'game-images' bucket
  dice_system public.dice_system not null default 'd20',
  starting_skill_points int not null default 0 check (starting_skill_points >= 0),
  skill_pip_cost int not null default 1 check (skill_pip_cost >= 0), -- per die, per pip
  main_pip_cost int not null default 10 check (main_pip_cost >= 0),
  players_can_edit_sheets boolean not null default false,
  dm_id uuid not null default auth.uid() references public.profiles on delete cascade,
  created_at timestamptz not null default now()
);

create table public.game_members (
  game_id bigint not null references public.games on delete cascade,
  user_id uuid not null references public.profiles on delete cascade,
  role public.member_role not null default 'player',
  status public.member_status not null default 'invited',
  last_seen_activity_at timestamptz not null default now(),
  created_at timestamptz not null default now(),
  primary key (game_id, user_id)
);

-- game_id null = default stat shared by every game
-- parent_id null = main stat; otherwise a skill under that main stat
create table public.stat_definitions (
  id bigint generated always as identity primary key,
  game_id bigint references public.games on delete cascade,
  parent_id bigint references public.stat_definitions on delete cascade,
  name text not null,
  description text,
  sort_order int not null default 0,
  created_at timestamptz not null default now()
);
create unique index stat_definitions_unique_name
  on public.stat_definitions (coalesce(game_id, 0), coalesce(parent_id, 0), lower(name));

create table public.characters (
  id bigint generated always as identity primary key,
  game_id bigint not null references public.games on delete cascade,
  owner_id uuid not null default auth.uid() references public.profiles on delete cascade,
  name text not null,
  species text,
  notes text,
  created_at timestamptz not null default now()
);

-- Main stat rows store the FULL value; skill rows store a BONUS above the
-- main stat. A skill with no row equals its main stat. (Rules: rules.py)
create table public.character_stats (
  character_id bigint not null references public.characters on delete cascade,
  stat_id bigint not null references public.stat_definitions on delete cascade,
  dice int not null default 0 check (dice >= 0),
  pips int not null default 0 check (pips between 0 and 2),
  value int not null default 0, -- used by D20 games (modifier)
  primary key (character_id, stat_id)
);

create table public.spells (
  id bigint generated always as identity primary key,
  game_id bigint not null references public.games on delete cascade,
  name text not null,
  description text,
  level int,
  stat_id bigint references public.stat_definitions on delete set null,
  created_at timestamptz not null default now(),
  unique (game_id, name)
);

create table public.character_spells (
  character_id bigint not null references public.characters on delete cascade,
  spell_id bigint not null references public.spells on delete cascade,
  primary key (character_id, spell_id)
);

create table public.inventory_items (
  id bigint generated always as identity primary key,
  character_id bigint not null references public.characters on delete cascade,
  name text not null,
  description text,
  quantity int not null default 1 check (quantity >= 0),
  created_at timestamptz not null default now()
);

create table public.game_notes (
  id bigint generated always as identity primary key,
  game_id bigint not null references public.games on delete cascade,
  title text,
  body text not null,
  visible_to_players boolean not null default false,
  created_at timestamptz not null default now()
);

-- Traits a character has or doesn't
create table public.abilities (
  id bigint generated always as identity primary key,
  game_id bigint not null references public.games on delete cascade,
  name text not null,
  description text,
  created_at timestamptz not null default now(),
  unique (game_id, name)
);

create table public.character_abilities (
  character_id bigint not null references public.characters on delete cascade,
  ability_id bigint not null references public.abilities on delete cascade,
  primary key (character_id, ability_id)
);

-- Append-only points ledger: + awards, − spends
create table public.point_transactions (
  id bigint generated always as identity primary key,
  character_id bigint not null references public.characters on delete cascade,
  amount int not null check (amount <> 0),
  reason text,
  created_by uuid default auth.uid() references public.profiles on delete set null,
  created_at timestamptz not null default now()
);

-- Written only by triggers; visible to all members of the game
create table public.game_activity (
  id bigint generated always as identity primary key,
  game_id bigint not null references public.games on delete cascade,
  character_id bigint references public.characters on delete cascade,
  actor_id uuid references public.profiles on delete set null,
  kind text not null, -- 'stat_change' | 'points' | 'stat_definition'
  source text not null, -- 'points' | 'direct' | 'gm' | 'award'
  message text not null,
  created_at timestamptz not null default now()
);

-- Indexes on foreign keys (keeps security checks fast)
create index on public.game_members (user_id);
create index on public.stat_definitions (game_id);
create index on public.stat_definitions (parent_id);
create index on public.characters (game_id);
create index on public.characters (owner_id);
create index on public.character_stats (stat_id);
create index on public.spells (game_id);
create index on public.character_spells (spell_id);
create index on public.inventory_items (character_id);
create index on public.game_notes (game_id);
create index on public.abilities (game_id);
create index on public.character_abilities (ability_id);
create index on public.point_transactions (character_id);
create index on public.game_activity (game_id, created_at desc);


-- 3. HELPER FUNCTIONS private schema: not callable through the API ____

create function private.is_game_member(gid bigint)
returns boolean language sql stable security definer set search_path = ''
as $$
  select exists (select 1 from public.game_members where game_id = gid and user_id = (select auth.uid()) and status = 'active');
$$;

create function private.has_game_invite_or_membership(gid bigint)
returns boolean language sql stable security definer set search_path = ''
as $$
  select exists (select 1 from public.game_members where game_id = gid and user_id = (select auth.uid()));
$$;

create function private.is_game_dm(gid bigint)
returns boolean language sql stable security definer set search_path = ''
as $$
  select exists (select 1 from public.games where id = gid and dm_id = (select auth.uid()));
$$;

-- Owner of the character, or DM of its game
create function private.can_manage_character(cid bigint)
returns boolean language sql stable security definer set search_path = ''
as $$
  select exists (select 1 from public.characters c
                 where c.id = cid
                   and (c.owner_id = (select auth.uid()) or private.is_game_dm(c.game_id)));
$$;

-- DM always; owner too when the game allows player editing
create function private.can_edit_character_stats(cid bigint)
returns boolean language sql stable security definer set search_path = ''
as $$
  select exists (select 1 from public.characters c
                 join public.games g on g.id = c.game_id
                 where c.id = cid
                   and (g.dm_id = (select auth.uid())
                        or (c.owner_id = (select auth.uid()) and g.players_can_edit_sheets)));
$$;

create function private.is_character_dm(cid bigint)
returns boolean language sql stable security definer set search_path = ''
as $$
  select exists (select 1 from public.characters c
                 where c.id = cid and private.is_game_dm(c.game_id));
$$;

create function private.stat_fits_character(sid bigint, cid bigint)
returns boolean language sql stable security definer set search_path = ''
as $$
  select exists (select 1 from public.stat_definitions s
                 join public.characters c on c.id = cid
                 where s.id = sid and (s.game_id is null or s.game_id = c.game_id));
$$;

create function private.spell_fits_character(spid bigint, cid bigint)
returns boolean language sql stable security definer set search_path = ''
as $$
  select exists (select 1 from public.spells s
                 join public.characters c on c.id = cid
                 where s.id = spid and s.game_id = c.game_id);
$$;

create function private.ability_fits_character(aid bigint, cid bigint)
returns boolean language sql stable security definer set search_path = ''
as $$
  select exists (select 1 from public.abilities a
                 join public.characters c on c.id = cid
                 where a.id = aid and a.game_id = c.game_id);
$$;

-- Game id from a storage path like '12/abc.png' (null if not numeric)
create function private.path_game_id(object_name text)
returns bigint language sql immutable set search_path = ''
as $$
  select case when split_part(object_name, '/', 1) ~ '^[0-9]+$'
              then split_part(object_name, '/', 1)::bigint end;
$$;

-- Plain formatting for log messages: '4D+2' (D6) or '+5' (D20)
create function private.fmt_stat(system public.dice_system, d int, p int, v int)
returns text language sql immutable set search_path = ''
as $$
  select case when system = 'd6'
              then coalesce(d, 0) || 'D' || case when coalesce(p, 0) > 0 then '+' || p else '' end
              else case when coalesce(v, 0) >= 0 then '+' else '' end || coalesce(v, 0) end;
$$;


-- 4. TRIGGERS _________________________________________________________
-- Profile for every new user (username from sign-up metadata)

create function private.handle_new_user()
returns trigger language plpgsql security definer set search_path = ''
as $$
begin
  insert into public.profiles (id, username)
  values (new.id, coalesce(new.raw_user_meta_data ->> 'username', 'player_' || substr(new.id::text, 1, 8)));
  return new;
end;
$$;

create trigger on_auth_user_created
  after insert on auth.users
  for each row execute function private.handle_new_user();

-- Game creator becomes its DM member
create function private.add_dm_as_member()
returns trigger language plpgsql security definer set search_path = ''
as $$
begin
  insert into public.game_members (game_id, user_id, role, status)
  values (new.id, new.dm_id, 'dm', 'active');
  return new;
end;
$$;

create trigger on_game_created
  after insert on public.games
  for each row execute function private.add_dm_as_member();

-- Stat hierarchy: two levels, and a skill's main stat must be default or same game
create function private.check_stat_parent()
returns trigger language plpgsql security definer set search_path = ''
as $$
declare
  p_game bigint;
  p_parent bigint;
begin
  if new.parent_id is null then
    return new;
  end if;
  select game_id, parent_id into p_game, p_parent
  from public.stat_definitions where id = new.parent_id;
  if p_parent is not null then
    raise exception 'Sub-stats can only be placed under a main stat';
  end if;
  if p_game is not null and p_game is distinct from new.game_id then
    raise exception 'Parent stat belongs to a different game';
  end if;
  return new;
end;
$$;

create trigger check_stat_parent
  before insert or update on public.stat_definitions
  for each row execute function private.check_stat_parent();

-- New characters receive the game's starting points
create function private.grant_starting_points()
returns trigger language plpgsql security definer set search_path = ''
as $$
declare
  pts int;
begin
  select starting_skill_points into pts from public.games where id = new.game_id;
  if coalesce(pts, 0) > 0 then
    insert into public.point_transactions (character_id, amount, reason, created_by)
    values (new.id, pts, 'Starting points', new.owner_id);
  end if;
  return new;
end;
$$;

create trigger on_character_created
  after insert on public.characters
  for each row execute function private.grant_starting_points();

-- ---------- Safety-net log for stat changes made outside the app ----------
-- Changes made through apply_stat_change() are logged there with the app's
-- description; this trigger catches anything else (e.g. dashboard edits).
create function private.log_stat_change()
returns trigger language plpgsql security definer set search_path = ''
as $$
declare
  v_cid bigint;
  v_sid bigint;
  v_char public.characters%rowtype;
  v_stat public.stat_definitions%rowtype;
  v_system public.dice_system;
  v_dm uuid;
  v_old text;
  v_new text;
  v_what text;
  v_change text;
  v_actor uuid := (select auth.uid());
  v_msg text;
  v_source text;
begin
  if nullif(current_setting('app.rpc_logging', true), '') is not null then
    return null; -- already logged by apply_stat_change()
  end if;

  if tg_op = 'DELETE' then
    v_cid := old.character_id; v_sid := old.stat_id;
  else
    v_cid := new.character_id; v_sid := new.stat_id;
  end if;

  select * into v_char from public.characters where id = v_cid;
  if not found then
    return null; -- character is being deleted
  end if;
  select * into v_stat from public.stat_definitions where id = v_sid;
  if not found then
    return null; -- stat definition is being deleted (logged separately)
  end if;
  select dice_system, dm_id into v_system, v_dm from public.games where id = v_char.game_id;

  if tg_op <> 'INSERT' then v_old := private.fmt_stat(v_system, old.dice, old.pips, old.value); end if;
  if tg_op <> 'DELETE' then v_new := private.fmt_stat(v_system, new.dice, new.pips, new.value); end if;
  if v_old is not distinct from v_new then
    return null;
  end if;

  -- Stored values: full value for main stats, bonus for skills
  v_what := v_stat.name || case when v_stat.parent_id is null then '' else ' bonus' end;
  v_change := case tg_op
    when 'INSERT' then 'set ' || v_what || ' to ' || v_new
    when 'DELETE' then 'cleared ' || v_what || ' (was ' || v_old || ')'
    else 'changed ' || v_what || ' from ' || v_old || ' to ' || v_new
  end;

  if v_actor = v_dm and v_actor is distinct from v_char.owner_id then
    v_msg := 'GM ' || v_change || ' for ' || v_char.name;
    v_source := 'gm';
  else
    v_msg := v_char.name || ' ' || v_change || ' (edited directly)';
    v_source := 'direct';
  end if;

  insert into public.game_activity (game_id, character_id, actor_id, kind, source, message)
  values (v_char.game_id, v_cid, v_actor, 'stat_change', v_source, v_msg);
  return null;
end;
$$;

create trigger log_stat_change
  after insert or update or delete on public.character_stats
  for each row execute function private.log_stat_change();

-- ---------- Log DM awards and corrections (spends are logged with their stat change) ----------
create function private.log_points()
returns trigger language plpgsql security definer set search_path = ''
as $$
declare
  v_name text;
  v_game bigint;
begin
  if nullif(current_setting('app.rpc_logging', true), '') is not null then
    return null;
  end if;
  if new.reason = 'Starting points' then
    return null;
  end if;

  select name, game_id into v_name, v_game from public.characters where id = new.character_id;

  insert into public.game_activity (game_id, character_id, actor_id, kind, source, message)
  values (v_game, new.character_id, new.created_by, 'points', 'award',
          case when new.amount > 0
               then 'GM awarded ' || new.amount || ' points to ' || v_name
               else 'GM deducted ' || abs(new.amount) || ' points from ' || v_name end
          || coalesce(': ' || new.reason, ''));
  return null;
end;
$$;

create trigger log_points
  after insert on public.point_transactions
  for each row execute function private.log_points();

-- Log when the GM adds, renames, moves or removes game-specific stats
create function private.log_stat_definition()
returns trigger language plpgsql security definer set search_path = ''
as $$
declare
  v_game bigint;
  v_parent text;
  v_msg text;
begin
  if tg_op = 'DELETE' then
    v_game := old.game_id;
  else
    v_game := new.game_id;
  end if;

  if v_game is null then
    return null; -- defaults are managed from the SQL Editor
  end if;
  if not exists (select 1 from public.games where id = v_game) then
    return null; -- whole game is being deleted
  end if;

  if tg_op = 'INSERT' then
    if new.parent_id is null then
      v_msg := 'GM added the main stat ' || new.name;
    else
      select name into v_parent from public.stat_definitions where id = new.parent_id;
      v_msg := 'GM added the skill ' || new.name || ' under ' || coalesce(v_parent, 'a main stat');
    end if;
  elsif tg_op = 'DELETE' then
    v_msg := 'GM removed the '
             || case when old.parent_id is null then 'main stat ' else 'skill ' end || old.name;
  else
    if new.name is distinct from old.name then
      v_msg := 'GM renamed ' || old.name || ' to ' || new.name;
    elsif new.parent_id is distinct from old.parent_id then
      select name into v_parent from public.stat_definitions where id = new.parent_id;
      v_msg := 'GM moved ' || new.name || ' under ' || coalesce(v_parent, 'another main stat');
    else
      return null;
    end if;
  end if;

  insert into public.game_activity (game_id, character_id, actor_id, kind, source, message)
  values (v_game, null, (select auth.uid()), 'stat_definition', 'gm', v_msg);
  return null;
end;
$$;

create trigger log_stat_definition
  after insert or update or delete on public.stat_definitions
  for each row execute function private.log_stat_definition();

-- 5. APPLYING STAT CHANGES ____________________________________________
-- game rules live in the app's rules.py; this function only guarantees integrity
-- ---------- Apply a stat change planned by the app ----------
-- Knows nothing about D6/D20 rules. It guarantees integrity:
--   * permissions: spending needs owner or GM; free edits need GM (or owner
--     when the game allows player editing)
--   * costs are never negative, balances never go below zero
--   * spending points can only raise a stat, never lower it
--   * the stat, the spend and the activity entry are saved together
-- Called from the app with sb.rpc('apply_stat_change', {...})
create function public.apply_stat_change(
  p_character_id bigint,
  p_stat_id bigint,
  p_dice int,
  p_pips int,
  p_value int,
  p_cost int,
  p_description text
)
returns jsonb language plpgsql security definer set search_path = ''
as $$
declare
  v_char public.characters%rowtype;
  v_actor uuid := (select auth.uid());
  v_is_dm boolean;
  v_old public.character_stats%rowtype;
  v_old_p int;
  v_new_p int;
  v_balance int;
  v_source text;
  v_msg text;
begin
  if p_cost is null or p_cost < 0 then
    raise exception 'Cost cannot be negative';
  end if;
  if coalesce(trim(p_description), '') = '' then
    raise exception 'A description of the change is required';
  end if;

  -- Lock the character so two changes can't spend the same points
  select * into v_char from public.characters where id = p_character_id for update;
  if not found then
    raise exception 'Character not found';
  end if;
  v_is_dm := private.is_game_dm(v_char.game_id);

  if p_cost > 0 then
    if v_char.owner_id is distinct from v_actor and not v_is_dm then
      raise exception 'You can only spend points on your own characters';
    end if;
  elsif not private.can_edit_character_stats(p_character_id) then
    raise exception 'Only the GM can change stats directly in this game';
  end if;

  if not private.stat_fits_character(p_stat_id, p_character_id) then
    raise exception 'That stat is not part of this game';
  end if;

  if p_cost > 0 then
    select * into v_old from public.character_stats
    where character_id = p_character_id and stat_id = p_stat_id;
    v_old_p := coalesce(v_old.dice, 0) * 3 + coalesce(v_old.pips, 0);
    v_new_p := coalesce(p_dice, 0) * 3 + coalesce(p_pips, 0);
    if v_new_p < v_old_p or coalesce(p_value, 0) < coalesce(v_old.value, 0)
       or (v_new_p = v_old_p and coalesce(p_value, 0) = coalesce(v_old.value, 0)) then
      raise exception 'Spending points can only raise a stat';
    end if;

    select coalesce(sum(amount), 0) into v_balance
    from public.point_transactions where character_id = p_character_id;
    if v_balance < p_cost then
      raise exception 'Not enough points: this costs %, you have %', p_cost, v_balance;
    end if;
  end if;

  perform set_config('app.rpc_logging', 'on', true);

  insert into public.character_stats (character_id, stat_id, dice, pips, value)
  values (p_character_id, p_stat_id, coalesce(p_dice, 0), coalesce(p_pips, 0), coalesce(p_value, 0))
  on conflict (character_id, stat_id)
  do update set dice = excluded.dice, pips = excluded.pips, value = excluded.value;

  if p_cost > 0 then
    insert into public.point_transactions (character_id, amount, reason, created_by)
    values (p_character_id, -p_cost, p_description, v_actor);
  end if;

  if p_cost > 0 then
    v_source := 'points';
    v_msg := v_char.name || ' ' || p_description || ' (spent ' || p_cost || ' points)';
  elsif v_is_dm and v_actor is distinct from v_char.owner_id then
    v_source := 'gm';
    v_msg := 'GM ' || p_description || ' for ' || v_char.name;
  else
    v_source := 'direct';
    v_msg := v_char.name || ' ' || p_description || ' (edited directly)';
  end if;

  insert into public.game_activity (game_id, character_id, actor_id, kind, source, message)
  values (v_char.game_id, p_character_id, v_actor, 'stat_change', v_source, v_msg);

  perform set_config('app.rpc_logging', '', true);

  select coalesce(sum(amount), 0) into v_balance
  from public.point_transactions where character_id = p_character_id;
  return jsonb_build_object('balance', v_balance);
end;
$$;

-- Profiles: logged-in users see usernames (to invite players)
create policy "Profiles readable" on public.profiles
  for select to authenticated using (true);
create policy "Update own profile" on public.profiles
  for update to authenticated
  using (id = (select auth.uid())) with check (id = (select auth.uid()));

-- Games
create policy "See own or joined games" on public.games
  for select to authenticated
  using (dm_id = (select auth.uid()) or private.has_game_invite_or_membership(id));
create policy "Create game as DM" on public.games
  for insert to authenticated with check (dm_id = (select auth.uid()));
create policy "DM updates game" on public.games
  for update to authenticated
  using (private.is_game_dm(id)) with check (dm_id = (select auth.uid()));
create policy "DM deletes game" on public.games
  for delete to authenticated using (private.is_game_dm(id));

-- Members
create policy "See members of my games" on public.game_members
  for select to authenticated
  using (user_id = (select auth.uid()) or private.is_game_member(game_id));
create policy "DM invites players" on public.game_members
  for insert to authenticated
  with check (private.is_game_dm(game_id) and role = 'player' and status = 'invited');
create policy "Accept own invite" on public.game_members
  for update to authenticated
  using (user_id = (select auth.uid()))
  with check (user_id = (select auth.uid()) and status = 'active');
create policy "DM removes players or player leaves" on public.game_members
  for delete to authenticated
  using (role = 'player' and (private.is_game_dm(game_id) or user_id = (select auth.uid())));

-- Stat definitions: defaults visible to all; game stats to members; DM edits own game's
create policy "See default and game stats" on public.stat_definitions
  for select to authenticated using (game_id is null or private.is_game_member(game_id));
create policy "DM adds game stats" on public.stat_definitions
  for insert to authenticated with check (game_id is not null and private.is_game_dm(game_id));
create policy "DM edits game stats" on public.stat_definitions
  for update to authenticated
  using (game_id is not null and private.is_game_dm(game_id))
  with check (game_id is not null and private.is_game_dm(game_id));
create policy "DM deletes game stats" on public.stat_definitions
  for delete to authenticated using (game_id is not null and private.is_game_dm(game_id));

-- Characters: owner and DM only
-- (to let the whole party see sheets: using (private.is_game_member(game_id)))
create policy "Owner or DM sees character" on public.characters
  for select to authenticated
  using (owner_id = (select auth.uid()) or private.is_game_dm(game_id));
create policy "Members create own characters" on public.characters
  for insert to authenticated
  with check (owner_id = (select auth.uid()) and private.is_game_member(game_id));
create policy "Owner or DM edits character" on public.characters
  for update to authenticated
  using (owner_id = (select auth.uid()) or private.is_game_dm(game_id))
  with check (private.is_game_member(game_id)
              and (owner_id = (select auth.uid()) or private.is_game_dm(game_id)));
create policy "Owner or DM deletes character" on public.characters
  for delete to authenticated
  using (owner_id = (select auth.uid()) or private.is_game_dm(game_id));

-- Character stats: owner + DM see; DM edits (owner too if the game allows)
create policy "View character stats" on public.character_stats
  for select to authenticated using (private.can_manage_character(character_id));
create policy "Permitted users add character stats" on public.character_stats
  for insert to authenticated
  with check (private.can_edit_character_stats(character_id)
              and private.stat_fits_character(stat_id, character_id));
create policy "Permitted users edit character stats" on public.character_stats
  for update to authenticated
  using (private.can_edit_character_stats(character_id))
  with check (private.can_edit_character_stats(character_id)
              and private.stat_fits_character(stat_id, character_id));
create policy "Permitted users remove character stats" on public.character_stats
  for delete to authenticated using (private.can_edit_character_stats(character_id));

-- Spells: members read, DM manages
create policy "Members see spells" on public.spells
  for select to authenticated using (private.is_game_member(game_id));
create policy "DM adds spells" on public.spells
  for insert to authenticated with check (private.is_game_dm(game_id));
create policy "DM edits spells" on public.spells
  for update to authenticated
  using (private.is_game_dm(game_id)) with check (private.is_game_dm(game_id));
create policy "DM deletes spells" on public.spells
  for delete to authenticated using (private.is_game_dm(game_id));

create policy "View character spells" on public.character_spells
  for select to authenticated using (private.can_manage_character(character_id));
create policy "Add character spells" on public.character_spells
  for insert to authenticated
  with check (private.can_manage_character(character_id)
              and private.spell_fits_character(spell_id, character_id));
create policy "Remove character spells" on public.character_spells
  for delete to authenticated using (private.can_manage_character(character_id));

-- Inventory: owner and DM
create policy "View inventory" on public.inventory_items
  for select to authenticated using (private.can_manage_character(character_id));
create policy "Add inventory" on public.inventory_items
  for insert to authenticated with check (private.can_manage_character(character_id));
create policy "Edit inventory" on public.inventory_items
  for update to authenticated
  using (private.can_manage_character(character_id))
  with check (private.can_manage_character(character_id));
create policy "Remove inventory" on public.inventory_items
  for delete to authenticated using (private.can_manage_character(character_id));

-- Notes: DM sees all, players see visible ones; DM writes
create policy "DM sees all notes, players see visible ones" on public.game_notes
  for select to authenticated
  using (private.is_game_dm(game_id) or (visible_to_players and private.is_game_member(game_id)));
create policy "DM adds notes" on public.game_notes
  for insert to authenticated with check (private.is_game_dm(game_id));
create policy "DM edits notes" on public.game_notes
  for update to authenticated
  using (private.is_game_dm(game_id)) with check (private.is_game_dm(game_id));
create policy "DM deletes notes" on public.game_notes
  for delete to authenticated using (private.is_game_dm(game_id));

-- Abilities: members read, DM manages; owner or DM picks for a character
create policy "Members see abilities" on public.abilities
  for select to authenticated using (private.is_game_member(game_id));
create policy "DM adds abilities" on public.abilities
  for insert to authenticated with check (private.is_game_dm(game_id));
create policy "DM edits abilities" on public.abilities
  for update to authenticated
  using (private.is_game_dm(game_id)) with check (private.is_game_dm(game_id));
create policy "DM deletes abilities" on public.abilities
  for delete to authenticated using (private.is_game_dm(game_id));

create policy "View character abilities" on public.character_abilities
  for select to authenticated using (private.can_manage_character(character_id));
create policy "Select character abilities" on public.character_abilities
  for insert to authenticated
  with check (private.can_manage_character(character_id)
              and private.ability_fits_character(ability_id, character_id));
create policy "Remove character abilities" on public.character_abilities
  for delete to authenticated using (private.can_manage_character(character_id));

-- Points ledger: owner + DM see; only DM inserts directly; no edits/deletes
create policy "Owner or DM sees point history" on public.point_transactions
  for select to authenticated using (private.can_manage_character(character_id));
create policy "DM awards points" on public.point_transactions
  for insert to authenticated with check (private.is_character_dm(character_id));

-- Activity: every member sees it; only triggers write it
create policy "Members see game activity" on public.game_activity
  for select to authenticated using (private.is_game_member(game_id));

  using (bucket_id = 'game-images'
         and private.has_game_invite_or_membership(private.path_game_id(name)));

  for insert to authenticated
  with check (bucket_id = 'game-images' and private.is_game_dm(private.path_game_id(name)));

  for delete to authenticated
  using (bucket_id = 'game-images' and private.is_game_dm(private.path_game_id(name)));


-- 10. DATA: profiles for any existing users, and default stats ________

insert into public.profiles (id, username)
select id, 'player_' || substr(id::text, 1, 8) from auth.users
on conflict (id) do nothing;

-- Default main stats and skills shared by every game (edit to taste)
with mains as (
  insert into public.stat_definitions (name, sort_order) values
    ('Strength', 1), ('Dexterity', 2), ('Intelligence', 3), ('Mechanical', 4), ('Technical', 5)
  returning id, name
)
insert into public.stat_definitions (parent_id, name, sort_order)
select m.id, s.name, s.ord
from mains m
join (values
  ('Strength', 'Running', 1),
  ('Strength', 'Climbing', 2),
  ('Strength', 'Lifting', 3),
  ('Strength', 'Jumping', 4),
  ('Strength', 'Swimming', 5),
  ('Dexterity', 'Dodge', 1),
  ('Dexterity', 'Stealth', 2),
  ('Dexterity', 'Sleight of Hand', 3),
  ('Intelligence', 'Lore', 1),
  ('Intelligence', 'Perception', 2),
  ('Mechanical', 'Repair', 1),
  ('Mechanical', 'Piloting', 2),
  ('Technical', 'Computers', 1),
  ('Technical', 'Medicine', 2)
) as s(main, name, ord) on s.main = m.name;
